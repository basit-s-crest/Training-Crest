# Python code to calculate the sum of the series 1 - 1/3 + 1/5 - 1/7 + ... for the first 10,000 terms and then multiplies the total by 4.

import math

def calculate_series_sum():
    terms = 10000
    sum_of_series = 0
    for i in range(terms):
        term = (-1)**i / (2*i + 1)
        sum_of_series += term
    result = sum_of_series * 4
    return result

result = calculate_series_sum()
print("The sum of the series is: ", result)

# Output:
# The sum of the series is: 3.141493536383679
